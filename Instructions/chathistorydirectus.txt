-- Tracks the last used sequence per year
CREATE TABLE emp_id_sequences (
  year INT PRIMARY KEY,
  last_seq INT NOT NULL DEFAULT 0,
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- Function to allocate the next EMPID like 20260001, 20260002, ...
CREATE OR REPLACE FUNCTION next_emp_id(p_year INT DEFAULT EXTRACT(YEAR FROM CURRENT_DATE)::INT)
RETURNS BIGINT
LANGUAGE plpgsql
AS $$
DECLARE
  v_seq INT;
  v_id BIGINT;
BEGIN
  INSERT INTO emp_id_sequences(year, last_seq)
  VALUES (p_year, 0)
  ON CONFLICT (year) DO NOTHING;

  UPDATE emp_id_sequences
     SET last_seq = last_seq + 1,
         updated_at = now()
   WHERE year = p_year
   RETURNING last_seq INTO v_seq;

  v_id := (p_year::BIGINT * 10000) + v_seq;  -- YYYY#### (#### up to 9999)
  RETURN v_id;
END;
$$;
